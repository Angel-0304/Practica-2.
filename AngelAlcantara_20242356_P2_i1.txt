FG-USER (FortiGate VM64-KVM v6.4.0)
=========================================================================

#Paso 0: acceso de gestion (unico paso por consola)

config system interface
    edit "port3"
        set mode static
        set ip 192.168.128.140 255.255.255.0
        set allowaccess ping https ssh http
    next
end

#Hostname 
config system global
    set hostname "FG-USER"
end

#Interfaces (GUI: Network > Interfaces)
config system interface
    edit "port1"
        set alias "WAN"
        set mode static
        set ip 20.24.23.2 255.255.255.252
        set allowaccess ping
        set role wan
    next
    edit "VLAN10"
        set vdom "root"
        set alias "USUARIOS"
        set interface "port2"
        set vlanid 10
        set ip 10.23.56.1 255.255.255.128
        set allowaccess ping
        set role lan
    next
end

#Servidor DHCP en VLAN10 (GUI: dentro de la interfaz VLAN10)
config system dhcp server
    edit 1
        set interface "VLAN10"
        set default-gateway 10.23.56.1
        set netmask 255.255.255.128
        set dns-service default
        config ip-range
            edit 1
                set start-ip 10.23.56.10
                set end-ip 10.23.56.100
            next
        end
    next
end

#Ruta por defecto hacia el ISP (GUI: Network > Static Routes)
config router static
    edit 1
        set gateway 20.24.23.1
        set device "port1"
    next
end

#Politica de salida con NAT (GUI: Policy & Objects > Firewall Policy)
config firewall policy
    edit 1
        set name "LAN-a-Internet"
        set srcintf "VLAN10"
        set dstintf "port1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
end

# ============================================================
# VPN SITE-TO-SITE (GUI: VPN > IPsec Wizard)
# ============================================================
config firewall address
    edit "VPN-A-SERVER_local_subnet_1"
        set subnet 10.23.56.0 255.255.255.128
    next
    edit "VPN-A-SERVER_remote_subnet_1"
        set subnet 10.23.56.128 255.255.255.240
    next
end

config firewall addrgrp
    edit "VPN-A-SERVER_local"
        set member "VPN-A-SERVER_local_subnet_1"
        set comment "VPN: VPN-A-SERVER (Created by VPN wizard)"
    next
    edit "VPN-A-SERVER_remote"
        set member "VPN-A-SERVER_remote_subnet_1"
        set comment "VPN: VPN-A-SERVER (Created by VPN wizard)"
    next
end

#Phase 1
# (Propuesta negociada: des-md5, limitada por la licencia de evaluacion)
config vpn ipsec phase1-interface
    edit "VPN-A-SERVER"
        set interface "port1"
        set peertype any
        set net-device disable
        set remote-gw 20.24.56.2
        set psksecret <PRE-SHARED-KEY>
        set comments "VPN: VPN-A-SERVER (Created by VPN wizard)"
    next
end

#Phase 2
config vpn ipsec phase2-interface
    edit "VPN-A-SERVER"
        set phase1name "VPN-A-SERVER"
        set src-addr-type name
        set dst-addr-type name
        set src-name "VPN-A-SERVER_local"
        set dst-name "VPN-A-SERVER_remote"
    next
end

#Ruta hacia la red remota por el tunel + ruta blackhole
config router static
    edit 2
        set device "VPN-A-SERVER"
        set dstaddr "VPN-A-SERVER_remote"
    next
    edit 3
        set distance 254
        set blackhole enable
        set dstaddr "VPN-A-SERVER_remote"
    next
end

# Politicas de la VPN (sin NAT)
config firewall policy
    edit 2
        set name "vpn_VPN-A-SERVER_local"
        set srcintf "VLAN10"
        set dstintf "VPN-A-SERVER"
        set srcaddr "VPN-A-SERVER_local"
        set dstaddr "VPN-A-SERVER_remote"
        set action accept
        set schedule "always"
        set service "ALL"
    next
    edit 3
        set name "vpn_VPN-A-SERVER_remote"
        set srcintf "VPN-A-SERVER"
        set dstintf "VLAN10"
        set srcaddr "VPN-A-SERVER_remote"
        set dstaddr "VPN-A-SERVER_local"
        set action accept
        set schedule "always"
        set service "ALL"
    next
end

#Red: /etc/netplan/01-lab.yaml
network:
  version: 2
  ethernets:
    eth0:
      addresses: [10.23.56.130/28]
      routes:
        - to: 0.0.0.0/0
          via: 10.23.56.129

#cloud-init: /etc/cloud/cloud.cfg.d/99-disable-network-config.cfg
network: {config: disabled}

#Paquetes instalados
apt install -y apache2 openssl curl traceroute

#Certificado autofirmado
openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
  -keyout /etc/ssl/private/lab.key \
  -out /etc/ssl/certs/lab.crt \
  -subj "/C=DO/ST=Santo Domingo/O=ITLA/OU=Lab VPN/CN=10.23.56.130" \
  -addext "subjectAltName=IP:10.23.56.130"

#Apache HTTPS: /etc/apache2/sites-available/default-ssl.conf
SSLCertificateFile      /etc/ssl/certs/lab.crt
SSLCertificateKeyFile   /etc/ssl/private/lab.key

a2enmod ssl
a2ensite default-ssl
systemctl restart apache2
systemctl enable apache2

#Redireccion HTTP -> HTTPS: /etc/apache2/sites-available/000-default.conf
Redirect permanent / https://10.23.56.130/

#Pruebas con el tunel activo (deben funcionar)
ping -c 4 10.23.56.130
traceroute -n 10.23.56.130
#  1  10.23.56.1
#  2  20.24.56.2
#  3  10.23.56.130
curl -k https://10.23.56.130

#Pruebas con el tunel deshabilitado (deben fallar)
ping -c 4 10.23.56.130                    # Destination Net Unreachable
traceroute -n 10.23.56.130                # 1  10.23.56.1 !N
curl -k --max-time 5 https://10.23.56.130 # Network is unreachable
